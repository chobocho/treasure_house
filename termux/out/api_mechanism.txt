== 1. termux-battery-status 의 몸통 ==
$ cat $PREFIX/bin/termux-battery-status
#!/data/data/com.termux/files/usr/bin/sh
set -e -u

SCRIPTNAME=termux-battery-status
show_usage () {
    echo "Usage: $SCRIPTNAME"
    echo "Get the status of the device battery."
    exit 0
}

while getopts :h option
do
    case "$option" in
        h) show_usage;;
        ?) echo "$SCRIPTNAME: illegal option -$OPTARG"; exit 1;
    esac
done
shift $((OPTIND-1))

if [ $# != 0 ]; then echo "$SCRIPTNAME: too many arguments"; exit 1; fi

/data/data/com.termux/files/usr/libexec/termux-api BatteryStatus
## tmx: side=termux cwd=/data/data/com.termux/files/home/github/treasure_house/termux exit=0

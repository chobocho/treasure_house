import hashlib, os, zlib
data = b"blob 6\0hello\n"            # 머리 + 몸
oid = hashlib.sha1(data).hexdigest()  # 이름 = 머리까지 넣은 SHA-1
d = ".git/objects/" + oid[:2]
os.makedirs(d, exist_ok=True)
with open(d + "/" + oid[2:], "wb") as f:
    f.write(zlib.compress(data))      # 파일 = zlib 으로 싼 바이트
print(oid)

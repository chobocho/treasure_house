#!/bin/sh
# clean: 표준 입력의 내용을 저장소 밖(../lfs-store)에 두고,
# 저장소에는 그 SHA-256 을 적은 세 줄짜리 "포인터" 만 넘긴다
mkdir -p ../lfs-store
tmp=$(mktemp)
cat > "$tmp"
oid=$(sha256sum "$tmp" | cut -c1-64)
mv "$tmp" ../lfs-store/$oid
printf 'pointer\noid sha256:%s\nsize %s\n' $oid $(wc -c < ../lfs-store/$oid)

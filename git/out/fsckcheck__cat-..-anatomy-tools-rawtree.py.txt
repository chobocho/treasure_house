import sys
# 트리 몸을 손으로 짠다: 항목마다 "<모드> <이름>\0" + 이름 20바이트.
# 인자: 모드 이름 blob40 … (적힌 차례 그대로 — git 처럼 정렬하지 않는다)
a = sys.argv[1:]
body = b""
for k in range(0, len(a), 3):
    body += ("%s %s" % (a[k], a[k + 1])).encode() + b"\0"
    body += bytes.fromhex(a[k + 2])
sys.stdout.buffer.write(body)

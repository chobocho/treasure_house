package fmaprobe

func Fused(w, f [8]float32) float32 {
	var s float32
	for i := 0; i < 8; i++ {
		s += w[i] * f[i]
	}
	return s
}

func Rounded(w, f [8]float32) float32 {
	var s float32
	for i := 0; i < 8; i++ {
		s += float32(w[i] * f[i])
	}
	return s
}
